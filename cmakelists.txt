cmake_minimum_required(VERSION 3.10)
project(consoleGame)

include(FetchContent)

# 1. Configure PDCursesMod variables BEFORE fetching
# Build statically to avoid .dll / .lib linkage issues in MSVC
set(PDC_BUILD_SHARED OFF CACHE BOOL "Build static library" FORCE) 
set(PDC_SDL2_BUILD OFF CACHE BOOL "Build SDL2 backend" FORCE)
set(PDC_GL_BUILD OFF CACHE BOOL "Build GL backend" FORCE)
set(PDC_WIDE ON CACHE BOOL "Enable wide characters" FORCE)
set(PDC_UTF8 ON CACHE BOOL "Enable UTF-8 support" FORCE)

# Prevent dependencies from trying to install to C:\Program Files
set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/install_dir" CACHE STRING "Local install prefix" FORCE)

# 2. Let FetchContent handle the downloading and subdirectory mapping safely
FetchContent_Declare(
    PDCursesMod
    URL https://github.com/Bill-Gray/PDCursesMod/archive/refs/heads/master.zip
)
FetchContent_Populate(PDCursesMod)

file(GLOB PDC_CORE_SRC   "${pdcursesmod_SOURCE_DIR}/pdcurses/*.c")
file(GLOB PDC_WINCON_SRC "${pdcursesmod_SOURCE_DIR}/wincon/*.c")

add_library(pdcurses STATIC ${PDC_CORE_SRC} ${PDC_WINCON_SRC})

include(FetchContent)
FetchContent_Declare(
    stb
    GIT_REPOSITORY https://github.com/nothings/stb.git
    GIT_TAG        master
)
FetchContent_MakeAvailable(stb)

# 3. Fetch nlohmann_json
FetchContent_Declare(
  nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(nlohmann_json)

# 4. C++ standaard instellen
if(MSVC)
    add_compile_options(/std:c++latest)
else()
    set(CMAKE_CXX_STANDARD 26 CACHE STRING "C++ standard")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON CACHE BOOL "Require C++ standard" FORCE)

# 5. Verzamel alle .cpp bestanden
add_executable(consoleGame
    src/activities.cpp
    src/combat.cpp
    src/consoleGame.cpp
    src/consoleGameData.cpp
    src/mainLoop.cpp
    src/setupAndUtility.cpp
    src/gameManager.cpp
    src/drawing.cpp
    src/definitions.cpp
)

# 6. Include mappen toevoegen
target_include_directories(consoleGame PUBLIC include)
# FetchContent populates pdcursesmod_SOURCE_DIR automatically
target_include_directories(pdcurses PUBLIC "${pdcursesmod_SOURCE_DIR}")
target_include_directories(${PROJECT_NAME} PRIVATE ${stb_SOURCE_DIR})
target_compile_definitions(pdcurses PUBLIC PDC_WIDE PDC_FORCE_UTF8)
target_link_libraries(pdcurses PUBLIC winmm)

# 7. Link libraries (pdcurses is now correctly recognized as a static target)
target_link_libraries(consoleGame PRIVATE nlohmann_json::nlohmann_json winmm pdcurses)

set_target_properties(consoleGame PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

# 8. Copy JSON files
set(JSON_FILES 
    JSON/abilities.json 
    JSON/characters.json 
    JSON/enemies.json 
    JSON/items.json 
    JSON/quests.json 
    JSON/locations.json
)

foreach(json_file IN LISTS JSON_FILES)
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/${json_file}" 
                   "${CMAKE_CURRENT_BINARY_DIR}/${json_file}" COPYONLY)
endforeach()

add_custom_command(TARGET consoleGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/graphics"
    "$<TARGET_FILE_DIR:consoleGame>/graphics"
)

add_custom_command(TARGET consoleGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/JSON"
    "$<TARGET_FILE_DIR:consoleGame>/JSON"
)
