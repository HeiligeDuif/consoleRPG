cmake_minimum_required(VERSION 3.10)
project(consoleGame)

include(FetchContent)

set(PDCURSES_ZIP "${CMAKE_BINARY_DIR}/pdcurses.zip")
set(PDCURSES_SRC_DIR "${CMAKE_BINARY_DIR}/PDCursesMod-master")

if(NOT EXISTS "${PDCURSES_SRC_DIR}")
    message(STATUS "Downloading PDCursesMod source...")
    file(DOWNLOAD
        "https://github.com/Bill-Gray/PDCursesMod/archive/refs/heads/master.zip"
        "${PDCURSES_ZIP}"
        STATUS download_status
    )
    list(GET download_status 0 status_code)
    if(NOT status_code EQUAL 0)
        message(FATAL_ERROR "Failed to download PDCursesMod source code.")
    endif()

    message(STATUS "Extracting PDCursesMod...")
    file(ARCHIVE_EXTRACT
        INPUT "${PDCURSES_ZIP}"
        DESTINATION "${CMAKE_BINARY_DIR}"
    )
endif()

file(GLOB PDCURSES_SOURCES 
    "${PDCURSES_SRC_DIR}/pdcurses/*.c"
    "${PDCURSES_SRC_DIR}/wincon/*.c"
)

FetchContent_Declare(
  nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)

FetchContent_MakeAvailable(nlohmann_json)

# C++ standaard instellen
if(MSVC)
    add_compile_options(/std:c++latest)
else()
    set(CMAKE_CXX_STANDARD 26)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Verzamel alle .cpp bestanden in de src map
add_executable(consoleGame
src/activities.cpp
src/combat.cpp
src/consoleGame.cpp
src/consoleGameData.cpp
src/mainLoop.cpp
src/setupAndUtility.cpp
src/gameManager.cpp
${PDCURSES_SOURCES})

# Include mappen toevoegen
target_include_directories(consoleGame PUBLIC include)
target_include_directories(consoleGame PRIVATE 
    "${PDCURSES_SRC_DIR}"
    "${PDCURSES_SRC_DIR}/pdcurses"
    "${PDCURSES_SRC_DIR}/wincon"
)
target_link_libraries(consoleGame PRIVATE nlohmann_json::nlohmann_json winmm)
set_target_properties(consoleGame PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

set(JSON_FILES JSON/abilities.json JSON/characters.json JSON/enemies.json JSON/items.json JSON/quests.json JSON/locations.json)

foreach(json_file ${JSON_FILES})
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/${json_file}" 
                   "${CMAKE_CURRENT_BINARY_DIR}/${json_file}" COPYONLY)
endforeach()

add_custom_command(TARGET consoleGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/graphics"
    "$<TARGET_FILE_DIR:consoleGame>/graphics"
)

add_custom_command(TARGET consoleGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/JSON"
    "$<TARGET_FILE_DIR:consoleGame>/JSON"
)
